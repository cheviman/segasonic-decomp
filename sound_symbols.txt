# SegaSonic The Hedgehog (MAME set 'sonic') - Z80 sound program symbols (epr-15785.ic36)
#
# Addresses and names only: no game data lives in this file.
# Format:  ADDRESS  KIND  NAME  [...]   [# comment]
#   func     a function entry
#   loop     the main loop's top: the program waits there for the board (a function entry)
#   code     a code label
#   jumps    NAME COUNT: a table of COUNT 16-bit code addresses
#   via      TABLE: the JP (HL), JP (IX) or JP (IY) at ADDRESS goes to an entry of TABLE
#   data     where a piece of data starts (the pieces of assets/sound/data, tools/data_formats.py)
# Addresses are the Z80's: 0000-9FFF is the start of the sound ROM, A000-BFFF a banked page of it,
# C000-DFFF the RF5C68, E000-FFFF the RAM shared with the V60 (V60 700000-701FFF). The data in the banked
# part has lines of its own, with the offset in the chip (bank * 2000h + the address in A000-BFFF - A000h):
#   OFFSET   banked NAME                 where a piece of data starts there (6 hex digits)
#
# Names for the decompiled C (tools/decompile_sound.py), as symbols.txt has them (disassemble.load_names):
#   ADDRESS  var    NAME TYPE [STRUCT]   a variable in the shared RAM (TYPE: u8 s8 u16 s16); a struct name
#                                        after the type: it holds the address of one
#   ADDRESS  table  NAME TYPE            data in the program ROM the code reads by index, TYPE per element
#   ADDRESS  io     NAME TYPE            a register of the RF5C68 (C000-CFFF) or a place of the sample
#                                        bank's window, read and written through the bus
#   ADDRESS  port   NAME                 an I/O port
#   struct   NAME   SIZE                 a structure in the shared RAM (SIZE in hexadecimal)
#   array    STRUCT FIRST COUNT          COUNT of them one after another from FIRST (COUNT in decimal)
#   field    STRUCT OFFSET NAME TYPE [STRUCT]  one of its fields
#   param    FUNCTION REGISTER NAME      what a function takes or hands back in a register (bc, de, hl, ix,
#                                        iy, a)
#   result   FUNCTION REGISTER NAME      what a function nothing calls hands back in a register (no
#                                        caller says it: the unused code at the end)
#   local    FUNCTION WHERE REGISTER NAME  the local variable that holds what the instructions at WHERE
#                                        (addresses separated by commas) write in REGISTER; "entry" is
#                                        what it holds when the function starts (build/sound/variables.txt
#                                        lists them: tools/decompile_sound.py --variables)
# build/sound/listing.s and build/sound/index.txt (written with the C) are what naming works from.
#
# FILE lines lay the functions out in the files of src/sound/ (tools/decompile.py source_files, as
# symbols.txt's):
#   ADDRESS  file   PATH   # what the stretch holds
# the functions from ADDRESS up to the next file line go to src/sound/PATH.c. A file may take several
# stretches; its functions go in the order of their addresses, and the comment of its first line is
# its description (the Description of the file's header).

field effect_header 01 first_status u8  # the sequence's first MIDI status byte: its low nibble the channel (A-E: players EC00-F000)
param read_table_word hl table table_words                         # every caller passes a table of the program ROM (instruments 193B, drum_kits 3F2E, part_templates 1594, vibrato_presets 45E2, pcm_sample_headers 6429, fm_patches 242E, playlist_command_table 4706, sound_groups 4A78, songs 4C66) or a sound_list's entries (48F6-48FA): the game's data keeps every index inside its table (programs 00-1D and F0-F2, samples up to AE, CC 20 up to 0C); the word 0405-0408 reads is the ROM's
local play_midi_message 0687 r2 pitch_entry pcm_pitch_steps            # pcm_pitch_tables[semitone] + 2 * (channel_pitch_bends[channel] + sample_detune) (067D-0687): an RF5C68 FD in the runs from 202E (pcm_pitch_tables' entries 202E-218E, +- the bend), read 0688-068B
local retune_voice 080C r2 pitch_entry pcm_pitch_steps                 # pcm_pitch_tables[semitone] + 2 * pitch_bend (07FF-080C), read 080D-0810
local retune_voice 1218 r2 fnum_entry fm_fnumbers                  # fm_fnum_tables[semitone] + 2 * pitch_bend (120B-1218): a YM3438 F-number in the runs from 13D4 (entries 13D4-1534), read 1219-121C
local play_midi_message 113C r2 fnum_entry rom        # the run + the bend;
local play_midi_message 113D r2 fnum_entry fm_fnumbers        # FM note on, fm_fnum_tables[semitone] + 2 * (channel_pitch_bends[channel] + fm_detune) (1133-113D), read 113E-1141
local step_fm_vibrato 6220 r2 fnumber fm_fnumbers                  # adds (to 'local step_fm_vibrato 006216,...,006414 hl fnumber'): fm_fnum_tables[note & 0F] + 2 * pitch_bend (6213-6220), read 6221-6224 into vibrato_center
local play_midi_message 115E r1 register_list fm_register_numbers          # DE as load_fm_patch hands it back, fm_level_registers (1360): the operators' total level registers (1164, INC DE 1166)
local play_midi_message 115E r2 patch_levels fm_patch_bytes           # HL as load_fm_patch hands it back, the patch's last 4 bytes (fm_patches[patch] + 15h): its operators' levels (1168, INC HL 1169)
local load_fm_patch 1332 r2 patch_data rom       # fm_patches;
local load_fm_patch 1335 r2 patch_data fm_patch_bytes       # the patch's bytes;
local load_fm_patch 1343 r2 patch_data       # new (the C calls it hl): fm_patches (1332), then the patch (read_table_word at 1335): 1342 reads its first 21 bytes, one a pass (INC HL 1343); handed back past them (the param line's patch_data)
local apply_fm_music_volume 12A9 r2 patch_levels rom  # fm_patches;
local apply_fm_music_volume 12AC r2 patch_levels fm_patch_bytes  # fm_patches, then the patch (12AC), + 15h (12AF-12B2): its 4 operator levels, read 12BD (INC HL 12BE)
local apply_fm_channel_volume 1301 r2 patch_levels rom    # fm_patches;
local apply_fm_channel_volume 1304 r2 patch_levels fm_patch_bytes    # the same for CC 7 (1304, 1307-130A, read 1315)
local playlist_command 46B8 r2 playlist playlist_bytes                # adds (the param line's playlist): F0 (playlist_jump), the 16-bit address after the command (46B5-46B8) where the playlist goes on; 46B9 reads the song there (the playlists and their jumps lie in 4D66-4E53)
field sequence_player 20 playlist u16 playlist_bytes                  # the next playlist byte, in the program ROM (the playlists 4D66-4E53 and their F0 jumps): step_player reads it at 4883-4889 (stored back at 4896); 48E0 sets it (492C) to the byte after the first song

# ======== what the C says above each function (build/naming/comments_1..8.txt) ========
comment chain_next_event Hands back data_byte without bit 7, which marks it as chained: the player's flags bit 6 is set so step_player reads the next event at once, with no delta time between them.
comment clear_player Zeroes the first 0x31 bytes of the sequence player at player, which stops it, and hands back its address.
comment clear_player_end_flag Clears player_end_flag; step_player calls it when a player with flags bit 4 stops. Nothing in the sound program sets that bit or reads player_end_flag.
comment fade_music_volume The music player's fade: each time music_fade_wait runs down (reloaded from music_fade_speed) its fade_volume drops by one and goes out as controller 0x10 on channel 0xE (BE 10 v); at 0 it stops the player, clears music_fade_speed and sends BE 50 7E and BE 79 00. Nothing in the sound program sets music_fade_speed.
comment fade_player_volume The fade of the sequence player at player: each time fade_wait runs down (reloaded from fade_speed) fade_channel_volume drops by one and goes out as controller 7 on channel 0xA (BA 07 v); at 0 it stops the player and sends BA 7B 00 (all notes off). Only while fade_speed is set, which nothing in the sound program does.
comment find_lowest_player Unused: of the byte at player_field and the same byte one and two pages on (the same field of the next two sequence players), hands back the address of the lowest, the first of equal ones, also kept in lowest_player_field.
comment next_sequence_byte Hands back cursor + 1; past the bank window (0xC000) it maps the player's next bank (bank + 1, out to ROM_BANK_LOW) and goes on from 0xA000.
comment playlist_command Playlist bytes F0-FF by their low nibble (playlist_command_table): F0, F3-FD jump to the address that follows, taking the song there; F1, F2 take song 0x90 + (playlist_song_choice_1/_2 & 7); FE clears the player, FF all eight (spare_players for queue page 0xE2, never passed). Hands back playlist and playlist_byte (the song, 0 stops).
comment poll_sound_command Takes the V60's sound command: once sound_command_group is set, clears it and finds sound_command in its list (sound_groups). Header flags bit 7 clear: music on player 0 (inline sequence, song header or playlist); set: player 2, or with bit 6 players 3-7 by the first event's MIDI channel, if its priority is not below the playing sound's.
comment select_channel_player Picks the player for a sound by the MIDI channel of its first event (the low nibble of the header's second byte): player's page plus that channel - 0xA, so channels 0xA-0xE take players 3-7 from player 3's page.
comment sequence_tick The sequencer's work on each timer tick: tick_countdown runs 4 to 1; when it wraps, players 1-7 that are playing are stepped (fade_player_volume on player 1 first), and when it reaches 2 the music player 0 is (fade_music_volume first while music_fade_speed is set). Nothing while midi_queue_count is 0x40 or more.
comment start_sequence Starts the sequence of the song header at header on the player at player: its bank into bank, its address into position, delta_time 1 (the first event at the next step) and flags bit 7 (playing).
comment start_song Starts song song_number of the songs table on the player at player (start_sequence).
comment step_player Counts delta_time down; at 0 copies the player's events (through the bank window) into the MIDI queue at midi_queue_write until one has a delta time after it: note offs get velocity 0x7F, controller 0x10 goes through fade_volume, sysex is skipped but counted. FF 2F or a status F1-F6/F8-FE ends it: playlist (flags bit 5) or stop.
comment step_player_if_playing Runs step_player for the player at player_address when its flags bit 7 (playing) is set.
comment allocate_sample_voice A sample note on for part: takes a free sample_voice, or keys off and takes the one with the lowest allocation_order (every order drops by 1; drum notes off channel 15 get orders from 0xF0, stolen last); sets its RF5C68 channel up, starts the sample and writes the pitch (pcm_pitch_tables, bend, detune and octave, or kit_pitch), the level and the pan; keys it on.
comment handle_midi_message Takes one message off the MIDI queue when midi_queue_count is not 0: its status byte, or under running status midi_status again, then play_midi_message for each of the 13 parts that is on and listens to its channel, each reading from the same place; then skip_midi_data moves past it.
comment init_midi_port Unused: sets the MIDI port's 8251 USART up: three 0 commands and an internal reset (0x40), mode 0x4E (asynchronous, clock / 16, 8 data bits, no parity, 1 stop bit: MIDI's frame), then command 0x37 as reset_midi_errors.
comment play_midi_message One MIDI message for part, by midi_status's high nibble: note on and off (FM voices, or sample voices for sample instruments and drum kits), pitch bend (retune_voice), control change (1, 7, 0x10, 0x11, 0x20, 0x40, 0x50, 0x51, 0x7B, 0x80) or program change; 0xA0 and 0xD0 only skip their data. The sample loops of CC 0x7B and CC 0x50 0x7E never leave the first voice.
comment note_off A note off for part (MIDI 0x80, or a note on with velocity 0): nothing for a drum kit (program 0xF0 and up); an FM instrument's goes to fm_note_off; else the sample voice on midi_status's channel playing note_code, if it is sounding and not already released, is stopped (stop_pcm_voice) unless the channel's sustain pedal is down (channel_controls bit 7), which only marks it released (flags bit 1).
comment read_midi_byte Moves midi_queue_read on by one and hands back the midi_queue byte there in byte (and in byte_copy's low byte) and its address in queue_slot.
comment receive_midi_byte Unused: one byte from the MIDI port into midi_queue: waits out (resetting) USART errors, reads without checking a byte has come, turns 0xF2 into 0x80 and drops the other 0xF0-0xFF bytes. A status byte opens a message of its length (3; 2 for 0xC0-0xDF), a lone data byte reopens the last (running status); a full message adds 1 to midi_queue_count.
comment receive_midi_byte_spare_return_1 Unused: an empty function, a lone return left after receive_midi_byte's own.
comment receive_midi_byte_spare_return_2 Unused: an empty function, a second lone return left after receive_midi_byte, before send_midi_byte.
comment reset_midi_errors Unused: command 0x37 to the MIDI USART (its only callers, receive_midi_byte and send_midi_byte, are unused too): resets its error flags and turns the transmitter, receiver, DTR and RTS on.
comment retune_voice Retunes a sounding voice to pitch_bend: FM (chip_channel bit 7), registers A4/A0 from fm_fnum_tables plus part's fm_detune; a sample voice, RF5C68 registers 3/2 from pcm_pitch_tables shifted by its octave, pitch_step and stream_position following. A sample voice of a nonzero octave uses up the caller's count: loop_count's high byte comes back 0.
comment scan_drum_note_voices Unused: a drum note on that, for MIDI note 0x2C (note_and_velocity's high byte), walks the 8 sample voices for one whose note is 0xCA but changes nothing whichever it finds; then, as for every other note, allocate_sample_voice.
comment send_midi_byte Unused: sends midi_byte out of the MIDI port once the USART is ready to transmit; an error it reports is reset (reset_midi_errors) and the wait starts over.
comment skip_midi_data Moves midi_queue_read to the last byte of the message that starts with first_byte: one data byte for 0xC0 and 0xD0 (program change, channel pressure), two for the others, after the status byte when first_byte is one.
comment update_fm_modulation CC 1 (modulation) for one sounding voice: an FM voice (chip_channel bit 7) gets register 0xB4 of its channel = 0xC0 (both outputs) | fm_pan_modulation's depth, as its frequency modulation sensitivity; a sample voice is left alone.
comment apply_fm_channel_volume CC 7 for one sounding FM voice of part, the same as apply_fm_music_volume does for CC 0x10: from the voice's level (level is the address of its chip_channel byte) and part's effective_volume a new note_level, added to the patch's operator levels that have bit 7 clear and written as their total levels.
comment apply_fm_music_volume CC 0x10 for one sounding FM voice of part (level is the address of the voice's chip_channel byte): note_level becomes 0x7F less half of the voice's level scaled by part's effective_volume (taken halfway to 0xFF); each of the 4 operator levels of its instrument's fm_patches entry with bit 7 clear is added to it and written as that operator's total level.
comment fm_note_off Note off for an FM program: the sounding fm_voice on midi_status's channel playing note_code, unless held already, is stopped (stop_fm_voice), or held (flags bit 1) while the channel's sustain pedal is down. de, saved_channel and a come back as the original's registers (&channel_controls[channel] after a match): CC 0x7B/0x80's loop walks on from them.
param fm_note_off de voice_size             # the voices' size the walk steps by, or the channel after a match (what CC 7B/80's loop goes on from)
param fm_note_off a channel                  # the channel the walk compared
comment load_fixed_fm_frequency Unused: hands back 0x3AE, read as registers A4/A0 take a frequency: block 0, F-number 0x3AE, a value of fm_fnum_tables.
comment load_fm_patch Writes the 21 registers of fm_patches[sample_or_patch] (their numbers from fm_patch_registers plus the channel within its part) to the YM3438 channel in fm_channel. Hands back channel_in_part (that channel number), register_list (now at fm_level_registers) and patch_data (at the patch's 4 operator levels) for the note on's level writes.
comment refill_pcm_voice Streams one sample voice a pass (refill_voice, the next of the 8 each time): a sounding one adds pitch_step to stream_position, and when that passes 0x800 copies 2 pages (256 bytes each) of its sample into its wave RAM, past 0x1000 4, wrapping the wave pages at 0xE0 and the bank window at 0xC000; a page ending in 0xFF ends the sample (stop_pcm_voice).
comment release_held_fm_voice The sustain pedal let go for an fm_voice held at its note off: flags and midi_channel 0 and its channel keyed off (register 0x28 of the YM3438 chip_channel bit 3 picks); unlike stop_fm_voice it leaves the vibrato fields as they are.
comment start_pcm_sample Starts sample sample_or_patch on voice: register 8 with only pcm_channels_on sounding, the sample's ROM bank from pcm_sample_headers (kept in the voice, set on the bank ports), its first 256 bytes copied to the voice's wave_page; wave_page and source_page move on (past the bank window, into the next bank). Header bytes 3-5 are kept, never read.
comment step_fm_vibrato One FM voice's software vibrato: vibrato_delay counts down first, then vibrato_wait (from vibrato_rate) between moves; each move takes vibrato_offset up by vibrato_rise or down by vibrato_fall (vibrato_phase bit 0) around vibrato_center, the note's F-number with pitch_bend, turning at the depth (vibrato_range, vibrato_depth), shifting vibrato_block; then registers A4/A0.
comment stop_fm_voice Frees the fm_voice voice (flags, midi_channel and its vibrato state 0) and keys its YM3438 channel off (register 0x28); hands back what write_fm_global does.
comment stop_pcm_voice Stops a sample voice: its RF5C68 channel's loop (register 5) to channel * 0x20 + 0x18, a stretch init_pcm_chip leaves silent; the voice freed (flags and midi_channel 0, wave_page 0xD0, stream_position 0x7FF), its bit out of pcm_channels_on, pcm_voices_used - 1, and the higher allocation_orders lowered (unless its own is 0x10 or more).
comment copy_sample_page Copies 256 bytes from sample_address to wave_address (a page of sample data into the RF5C68's wave memory window) and hands both back moved on by 256.
comment init_fm_chips Sets up both YM3438: the 11 register/value pairs of fm_global_init to part I of each (LFO, timers, every key off, DAC off), the first YM3438's timer A to 0x3EF (registers 0x24 = 0xFB, 0x25 = 3), then the 36 pairs of fm_operator_init to all four parts (total level 0x7F, sustain level and release 0xFF, SSG-EG 0 on every operator).
comment init_midi_parts Starts the 13 MIDI parts: part_templates[1]'s 8 parts (0x48 bytes) to MIDI part 0 and the first 0x2D bytes of part_templates[0] (5 parts) to part 8; then each part's effective_volume = volume * (music_volume, or 0xFF for parts with flags bit 0) >> 8, + 1 when not 0.
comment init_pcm_chip Clears the RF5C68's 16 wave banks through the window at 0xD000 (register 7 picks the bank, sound off): even banks to 0x80, odd ones to 0xFF but 0x2BC-0xABC to 0x80; each fill also writes the byte past the window (midi_in_message_length). Then the 41 register/value pairs of pcm_register_init.
comment init_pcm_tracks Starts the 8 sample voices: wave_page 0xD0 and stream_position 0x07FF each.
comment init_track_channels Sets the chip_channel of the 20 voices (the 8 sample voices, then the 12 FM voices after them) from voice_chip_channels: 0-7 the RF5C68's channels, 0x80-0x8E the YM3438's (bit 3 the chip, bit 2 the part).
comment multiply_bytes Hands back value's high byte times factor's low byte, 8 by 8 bits by shift and add (the volume scalings keep the product's high byte).
comment read_table_word Hands back the 16-bit entry index of the table at table.
comment sound_init Sets up the sound program: init_pcm_chip, init_fm_chips, the MIDI queue (midi_queue_read 0xFF, midi_queue_write 0, midi_status 0x80), tick_countdown 1, music_volume 0xFF, the voice cursors at their first voices, init_track_channels, init_pcm_tracks and init_midi_parts; then timer A started on both YM3438 (register 0x27 = 0x15).
comment sound_irq The timer interrupt (the first YM3438's timer A): acknowledges every pending input, rearms the timer (register 0x27 = 0x15) and runs sequence_tick, the sequence players' tick.
comment sound_main_loop One pass of the sound program's main loop: waits while sound_cpu_hold is set; when the second YM3438's timer A has run out, rearms it and refills a sample voice (refill_pcm_voice); step_fm_vibrato on vibrato_voice when its vibrato_depth is set, vibrato_voice moving on to the next of the 12 FM voices; poll_sound_command, handle_midi_message.
comment sound_reset The sound CPU's start: interrupts off, the shared RAM 0xE000-0xFEFF cleared, the interrupt inputs mapped (ports 0xD0-0xD2), unmasked and their pending ones cleared, sound_init, then interrupts on.
comment write_fm1 Writes value to register fm_register of the first YM3438's part I (ports 0x80 and 0x81), interrupts off between the two writes.
comment write_fm1_part2 Writes value to register fm_register of the first YM3438's part II (ports 0x82 and 0x83), interrupts off between the two writes.
comment write_fm2 Writes value to register fm_register of the second YM3438's part I (ports 0x90 and 0x91), interrupts off between the two writes.
comment write_fm2_part2 Writes value to register fm_register of the second YM3438's part II (ports 0x92 and 0x93), interrupts off between the two writes.
comment write_fm_channel Writes value to register fm_register on the chip and part of fm_channel's channel: the second YM3438 when its bit 3 is set, part II when its bit 2 is set.
comment write_fm_global Writes value to register fm_register on part I of the YM3438 that fm_channel's bit 3 picks (the second when set), for the registers of the whole chip such as key on/off (0x28); hands back value.
comment write_pcm_register Writes register_value to RF5C68 register pcm_register (at 0xC000 + it).
